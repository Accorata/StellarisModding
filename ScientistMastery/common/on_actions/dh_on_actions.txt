# No scope, like on_game_start
on_decade_pulse = {
	events = {
		dangerous_horizons.102			# Reaching End Game Year increases Extradimensional Scouting
	}
}

# this = country
on_decade_pulse_country = {
	events = {
		dangerous_horizons.1			# Random leader events
	}
}

# THIS = Ship
on_jump_drive = {
	events = {
		dangerous_horizons.101			# Using Jump Drives increases Extradimensional Scouting
	}
}

# Executed as a leader has died
# This = Country
# From = Leader
on_leader_death = {
	events = {
		dangerous_horizons.10          # Imperial Feedback government harvests Mastery traits
	}
}

# A country has increased the level of a tech, use last_increased_tech trigger to check tech and level.
# This = Country
on_tech_increased = {
	events = {
		dangerous_horizons.5			# Grant 5% of mega engineering for Dreams Beyond Scale
		dangerous_horizons.6			# Grant 5% of existential campaigns for Unending Ambition
		dangerous_horizons.100			# Researching Jump Drives starts Extradimensional Scouting
	}
}
