	events = {
		dangerous_horizons.5			# Grant 5% of mega engineering for Dreams Beyond Scale
		dangerous_horizons.6			# Grant 5% of existential campaigns for Unending Ambition
		dangerous_horizons.100			# Researching Jump Drives starts Extradimensional Scouting
	}
}
