building_dh_conduction_nexus = {
	base_buildtime = @b2_time
	category = resource
	empire_limit = {
		base = 1
	}

	building_sets = {
		generator
	}

	allow = {
		inline_script = {
			script = buildings/dh_has_upgraded_capital
		}
	}
	destroy_trigger = {
		OR = {
			has_modifier = resort_colony
			AND = {
				owner = {
					is_ai = yes
				}
				has_any_generator_district_or_building = no
			}
		}
	}

	potential = {
		NOR = {
			has_modifier = resort_colony
		}
		OR = {
			NOT = { is_planet_class = pc_habitat }
			AND = {
				is_planet_class = pc_habitat
				check_modifier_value = {
					modifier = district_hab_energy_max_add
					value > 0
				}
			}
		}
		OR = {
			owner = {
				is_ai = no
			}
			has_any_generator_district_or_building = yes
		}
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = rare_crystals
			COST = @b2_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		cost = {
			rare_crystals = @b2_rare_cost
		}
		upkeep = {
			energy = @b1_upkeep
			rare_crystals = @b2_rare_upkeep
		}
	}

	planet_modifier = {
		planet_technician_physics_research_produces_add = 0.5
	}

	inline_script = {
		script = jobs/technicians_add
		AMOUNT = 100
	}

	prerequisites = {
		tech_dh_conduction_nexus
	}
	show_tech_unlock_if = {
		is_wilderness_empire = no
	}
}
