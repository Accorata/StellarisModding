dh_create_scouting_fleet = {
	event_target:dh_extradimensional_scouts = {
		random_list = {
			15 = {
				create_leader = {
					class = commander
					species = event_target:extradimensional_species
					name = random
					skill = 3
					traits = {
						trait = leader_trait_ethereal
					}
				}
			}
			15 = {
				create_leader = {
					class = commander
					species = event_target:extradimensional_species
					name = random
					skill = 3
					traits = {
						trait = leader_trait_dimensional_stutter
					}
				}
			}
			30 = {
				create_leader = {
					class = commander
					species = event_target:extradimensional_species
					name = random
					skill = 3
				}
			}
			40 = {
				create_leader = {
					class = commander
					species = event_target:extradimensional_species
					name = random
					skill = 3
					traits = {
						trait = leader_trait_dh_disintegrating
					}
				}
			}
		}
		create_fleet = {
			effect = {
				set_owner = PREV
				create_ship = {
					name = random
					design = "NAME_Revenant"
					graphical_culture = "extra_dimensional_01"
				}
				assign_leader = last_created_leader
				create_ship = {
					name = random
					design = "NAME_Revenant"
					graphical_culture = "extra_dimensional_01"
				}
				while = {
					count = 3
					create_ship = {
						name = random
						design = "NAME_Phantom"
						graphical_culture = "extra_dimensional_01"
					}
				}
				while = {
					count = 6
					create_ship = {
						name = random
						design = "NAME_Wraith"
						graphical_culture = "extra_dimensional_01"
					}
				}
				set_location = {
					target = ROOT
					distance = 2
					angle = random
				}
				set_fleet_stance = aggressive
				set_aggro_range = 200
				set_aggro_range_measure_from = self
				set_fleet_bombardment_stance = armageddon
			}
		}
	}
