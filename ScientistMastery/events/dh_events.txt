# Fires from on_decade_pulse_country
country_event = {
	id = dangerous_horizons.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_country_type = default
		NOT = {
			has_country_flag = dh_upgraded_to_mastery
		}
	}

	immediate = {
		country_event = {
			id = dangerous_horizons.2
			days = 1
			random = 1800
		}
	}
}

country_event = {
	id = dangerous_horizons.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_owned_leader = {
			NOT = { has_leader_flag = beholder }
			has_base_skill >= 3
			leader_years_of_service >= 10
			dh_has_level_3_expertise_trait = yes
		}
	}

	immediate = {
		random_owned_leader = {
			limit = {
				NOT = { has_leader_flag = beholder }
				has_base_skill >= 3
				leader_years_of_service >= 10
				dh_has_level_3_expertise_trait = yes
			}
			random_list = {
				15 = { leader_event = { id = dangerous_horizons.3 } }
				85 = { }
			}
		}
	}
}
