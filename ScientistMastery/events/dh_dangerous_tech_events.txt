## Fallen Empire Buildings ##

country_event = {
	id = dangerous_horizons.1000
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_fallen_empire = no
	}

	immediate = {
		# Energy
		if = {
			limit = {
				OR = {
					has_technology = tech_fe_singularity_1
					has_country_flag = class_3_singularity_can_build
				}
			}
			random_list = {
				90 = {}
				# Unstable Singularity
				10 = {
					modifier = {
						factor = 0
						OR = {
							has_country_flag = dh_unstable_singularity_event
							NOT = {
								any_owned_planet = {
									OR = {
										has_active_building = building_class_3_singularity
										has_active_building = building_class_4_singularity
									}
								}
							}
						}
					}
					modifier = {
						factor = value:dh_num_fallen_empire_buildings|BUILDING|building_class_3_singularity|BUILDING_2|building_class_4_singularity|
					}
					random_owned_planet = {
						limit = {
							OR = {
								has_active_building = building_class_3_singularity
								has_active_building = building_class_4_singularity
							}
						}
						carrier_event = {
							id = dangerous_horizons.1100
							days = 1
							random = 360
						}
					}
				}
			}
		}
	}
}

# Rampaging Invaders

carrier_event = {
	id = dangerous_horizons.1200
	title = dangerous_horizons.1200.name
	desc = dangerous_horizons.1200.desc
	picture = GFX_evt_alien_nature # change...
	show_sound = event_air_raid_siren # change...
	is_triggered_only = yes
	fire_only_once = yes

	event_chain = dh_rampaging_invaders_chain

	trigger = {
		owner = {
			NOT = {
				has_country_flag = dh_rampaging_invaders_event
			}
		}
		OR = {
			has_active_building = building_dimensional_replicator
			has_active_building = building_dimensional_fabricator
		}
	}
	
	immediate = {
		owner = {
			set_country_flag = dh_rampaging_invaders_event
		}
		set_planet_flag = dh_rampaging_invasion
		create_country = {
			name = "NAME_DH_Rampaging_Invaders"
			type = faction
			effect = {
				establish_communications_no_message = root.owner
				establish_contact = { who = root.owner location = root.solar_system }
				save_event_target_as = dh_rampaging_invaders_country
			}
		}
		create_species = {
			name = "NAME_DH_Extradimensional"
			class = EXD
			namelist = Extradimensional
			portrait = exd3_leader
			traits = random
			effect = { save_event_target_as = dh_rampaging_invaders_species }
		}
	}

	option = {
		name = dangerous_horizons.1200.a
		owner = {
			begin_event_chain = {
				event_chain = dh_rampaging_invaders_chain
				target = root
			}
		}
		while = {
			count = 8
			create_army = {
				name = "NAME_DH_Rampaging_Invaders"
				owner = dh_rampaging_invaders_country
				species = event_target:dh_rampaging_invaders_species
				type = "dh_extradimensional_invader_army"
			}
		}
	}

	option = {
		name = dangerous_horizons.1200.b
		owner = {
			begin_event_chain = {
				event_chain = dh_rampaging_invaders_chain
				target = root
			}
		}
		if = {
			limit = {
				has_active_building = building_dimensional_replicator
			}
			remove_building = building_dimensional_replicator
		}
		if = {
			limit = {
				has_active_building = building_dimensional_fabricator
			}
			remove_building = building_dimensional_fabricator
		}
		while = {
			count = 3
			create_army = {
				name = "NAME_DH_Rampaging_Invaders"
				owner = dh_rampaging_invaders_country
				species = event_target:dh_rampaging_invaders_species
				type = "dh_extradimensional_invader_army"
			}
		}
	}

	after = {
		owner = {
			country_event = {
				id = dangerous_horizons.1201
				days = 5
				random = 5
			}
		}
	}
}
